def algebra():
 # Route A: closed H rows inserted in the differentiated radial identities.
 def hw(j):
  if j==0:return {0:F(2,8),2:F(-1,8)}
  if j==1:return {0:F(2,8),1:F(1,8),2:F(-1,8),3:F(-1,8)}
  return {i:F(w,8) for i,w in [(j-1,1),(j,1),(j+1,-1),(j+2,-1)] if 0<=i<260}
 # Route B: signed sine scattering followed by angular raising/lowering.
 signed=[defaultdict(Q) for _ in range(262)]
 for j in range(260):
  for f in [1+2*j,1-2*j,3+2*j,3-2*j]:signed[(abs(f)-1)//2][j]+=Q(1 if f>0 else -1,8)
 for n in range(1,262):
  m=2*n
  for k in range(8):
   a=defaultdict(F);b=defaultdict(Q)
   for j,w in [(n-1,4-m),(n,4+m)]:
    for mode,v in hw(j).items():
     a[k,mode]+=w*v/2
     if k<7:a[k+1,mode]+=(k+1)*v/2
    for mode,v in signed[j].items():
     b[k,mode]+=Q(w,2)*v
     if k<7:b[k+1,mode]+=Q(k+1,2)*v
   for key in set(a)|set(b):
    need(Q(a[key].numerator,a[key].denominator)==b[key],'forcing symbolic coefficient');count['forcing_weight_comparisons']+=1
    digest.update((str((n,k,key))+'='+str(a[key])+'\n').encode())
   count['forcing_forms']+=1
  # Direct jet differentiation versus separately prescribed radial equations.
  # Coordinates are I,J,Hlower,Hupper,Hlower_prime,Hupper_prime.
  Ip=[F(-m-4),0,1,0,0,0];Jp=[0,F(m-4),0,-1,0,0]
  A=[F(-m),0,F(1,2),F(1,2),0,0];Bb=[0,F(m),F(-1,2),F(-1,2),0,0]
  Ap=[4*x-m*y for x,y in zip(A,Ip)];Bp=[4*x+m*y for x,y in zip(Bb,Jp)]
  Ap[4]+=F(1,2);Ap[5]+=F(1,2);Bp[4]-=F(1,2);Bp[5]-=F(1,2)
  FF=[0,0,Q(4-m,2),Q(4+m,2),Q(1,2),Q(1,2)]
  AQ=[Q(x.numerator,x.denominator) if isinstance(x,F) else Q(x) for x in A];BQ=[Q(x.numerator,x.denominator) if isinstance(x,F) else Q(x) for x in Bb]
  for i in range(6):
   need(Q(Ap[i].numerator,Ap[i].denominator)==-m*AQ[i]+FF[i],'A ODE')
   need(Q(Bp[i].numerator,Bp[i].denominator)==m*BQ[i]-FF[i],'B ODE')
   count['canonical_ODE_coefficients']+=2
  for i in range(6):
   expected=Q(-m) if i==0 else Q(m) if i==1 else Q(0)
   need(AQ[i]+BQ[i]==expected,'A+B=2mpsi');count['canonical_sum_coefficients']+=1
   dd=4*(Q(1) if i==1 else Q(-1) if i==0 else Q(0))+(Q(Jp[i].numerator,Jp[i].denominator) if isinstance(Jp[i],F) else Q(Jp[i]))-(Q(Ip[i].numerator,Ip[i].denominator) if isinstance(Ip[i],F) else Q(Ip[i]))
   need(BQ[i]-AQ[i]==dd,'B-A=2rpsi_r');count['canonical_difference_coefficients']+=1
 count['canonical_modes']=261
